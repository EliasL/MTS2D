# Homebrew
eval "$(/opt/homebrew/bin/brew shellenv)"

# Prefer Apple Clang over Homebrew Clang
export PATH="/usr/bin:/usr/sbin:/bin:/sbin:$PATH"

# OpenMP for Apple Clang (Homebrew libomp)
export CPPFLAGS="-Xpreprocessor -fopenmp -I/opt/homebrew/opt/libomp/include $CPPFLAGS"
export LDFLAGS="-lomp -L/opt/homebrew/opt/libomp/lib $LDFLAGS"
export LIBRARY_PATH="/opt/homebrew/opt/libomp/lib:$LIBRARY_PATH"

# Apple Clang
export CC=/usr/bin/clang
export CXX=/usr/bin/clang++
